cmake_minimum_required(VERSION 4.3)
project(shell LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(ProcessorCount)
if(NOT DEFINED CPU_COUNT)
	ProcessorCount(CPU_COUNT)
	message(WARNING "It's recommended to configure a maximum CPU count. The CPU count in your environment is ${CPU_COUNT}")
endif()
add_definitions(-DPROCESSOR_COUNT=${CPU_COUNT})
file(GLOB SOURCES *.cpp)
add_executable(server.out ${SOURCES})
